cmake_minimum_required(VERSION 3.21)
project(LeoCADReadbackTests LANGUAGES CXX)
find_package(Qt6 6.5 REQUIRED COMPONENTS Gui OpenGL Test)
set(CMAKE_AUTOMOC ON)
qt_add_executable(leocad-readback-tests tst_readback.cpp)
target_compile_features(leocad-readback-tests PRIVATE cxx_std_17)
target_include_directories(leocad-readback-tests PRIVATE ../../common)
target_link_libraries(leocad-readback-tests PRIVATE Qt6::Gui Qt6::OpenGL Qt6::Test)
set_target_properties(leocad-readback-tests PROPERTIES MACOSX_BUNDLE FALSE)
